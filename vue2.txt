<script setup>
import { ref, onMounted } from 'vue'
import ProductForm from './components/ProductForm.vue'
import ProductList from './components/ProductList.vue'

const API = 'http://localhost:8080/api/products'

const products = ref([])
const selected = ref(null)
const formKey = ref(0)
const message = ref('')
const error = ref('')

// READ ALL
async function getProducts() {
  const response = await fetch(API)
  if (!response.ok) throw new Error('Failed to fetch products')
  return response.json()
}

// READ ONE
async function getProductById(id) {
  const response = await fetch(`${API}/${id}`)
  if (!response.ok) throw new Error('Product not found')
  return response.json()
}

// CREATE
async function addProduct(product) {
  const response = await fetch(API, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(product)
  })

  if (!response.ok) throw new Error('Failed to add product')
  return response.json()
}

// UPDATE
async function updateProduct(id, product) {
  const response = await fetch(`${API}/${id}`, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(product)
  })

  if (!response.ok) throw new Error('Failed to update product')
  return response.json()
}

// DELETE
async function deleteProduct(id) {
  const response = await fetch(`${API}/${id}`, {
    method: 'DELETE'
  })

  if (!response.ok) throw new Error('Failed to delete product')
}

// Load products
async function loadProducts() {
  try {
    products.value = await getProducts()
    error.value = ''
  } catch (e) {
    error.value = e.message
  }
}

// CREATE and UPDATE
async function saveProduct(product) {
  try {
    if (selected.value) {
      await updateProduct(selected.value.id, product)
      message.value = 'Product updated!'
    } else {
      await addProduct(product)
      message.value = 'Product added!'
    }

    selected.value = null
    formKey.value++
    await loadProducts()
  } catch (e) {
    error.value = e.message
  }
}

// EDIT
function editProduct(product) {
  selected.value = { ...product }
  formKey.value++
}

// DELETE
async function removeProduct(product) {
  if (!confirm('Delete ' + product.name + '?')) return

  try {
    await deleteProduct(product.id)

    products.value = products.value.filter(
      p => p.id !== product.id
    )

    message.value = 'Product deleted!'
  } catch (e) {
    error.value = e.message
  }
}

// CANCEL EDIT
function cancelEdit() {
  selected.value = null
  formKey.value++
}

// Run when page opens
onMounted(loadProducts)
</script>

<template>
  <header>
    <h1>Product Manager</h1>
    <p>Product Inventory Management</p>
  </header>

  <main>
    <p v-if="message" class="success">{{ message }}</p>
    <p v-if="error" class="error">{{ error }}</p>

    <div class="layout">
      <ProductForm
        :key="formKey"
        :product="selected"
        @save="saveProduct"
        @cancel="cancelEdit"
      />

      <ProductList
        :products="products"
        @edit="editProduct"
        @delete="removeProduct"
      />
    </div>
  </main>
</template>

  const editing = selected.value !== null

<script setup>
import { ref } from 'vue'

const props = defineProps({
  product: Object
})

const emit = defineEmits(['save', 'cancel'])

function emptyForm() {
  return {
    name: '',
    category: 'Electronics',
    price: '',
    stock: ''
  }
}

const form = ref(
  props.product ? { ...props.product } : emptyForm()
)

function submit() {
  emit('save', {
    ...form.value,
    name: form.value.name.trim(),
    price: Number(form.value.price),
    stock: Number(form.value.stock)
  })
}
</script>

<template>
  <section class="panel">
    <h2>{{ props.product ? 'Edit Product' : 'Add Product' }}</h2>

    <form @submit.prevent="submit">
      <label>Name</label>
      <input v-model="form.name" required />

      <label>Category</label>
      <select v-model="form.category">
        <option>Electronics</option>
        <option>Accessories</option>
        <option>Stationery</option>
        <option>Other</option>
      </select>

      <label>Price</label>
      <input
        type="number"
        v-model="form.price"
        min="0"
        step="0.01"
        required
      />

      <label>Stock</label>
      <input
        type="number"
        v-model="form.stock"
        min="0"
        step="1"
        required
      />

      <button class="primary" type="submit">
        {{ props.product ? 'Update' : 'Add Product' }}
      </button>

      <button
        v-if="props.product"
        class="secondary"
        type="button"
        @click="emit('cancel')"
      >
        Cancel
      </button>
    </form>
  </section>
</template>
