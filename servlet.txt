import java.io.*;
import java.sql.*;
import javax.servlet.*;
import javax.servlet.http.*;

public class EmployeeServlet extends HttpServlet {

    public void doGet(HttpServletRequest request,
                      HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("text/html");
        PrintWriter out = response.getWriter();

        try {

            // 1. Load JDBC driver
            Class.forName("com.mysql.jdbc.Driver");

            // 2. Establish database connection
            Connection con =
                DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/TEST",
                    "root",
                    "password"
                );

            // 3. Create statement
            Statement stmt = con.createStatement();

            // 4. Retrieve employees whose age is > 45
            String selectSQL =
                "SELECT id, age, first, last, experience " +
                "FROM Employees WHERE age > 45";

            ResultSet rs = stmt.executeQuery(selectSQL);

            out.println("<html><body>");
            out.println("<h2>Updated Employee Details</h2>");

            // 5. Process each employee
            while (rs.next()) {

                int id = rs.getInt("id");
                int age = rs.getInt("age");
                String first = rs.getString("first");
                String last = rs.getString("last");

                // 6. Update experience to 10 years
                String updateSQL =
                    "UPDATE Employees SET experience = 10 " +
                    "WHERE id = " + id;

                stmt.executeUpdate(updateSQL);

                // 7. Display updated information
                out.println("ID: " + id + "<br>");
                out.println("Name: " + first + " " + last + "<br>");
                out.println("Age: " + age + "<br>");
                out.println("Job Experience: 10 years<br><br>");
            }

            out.println("</body></html>");

            // 8. Close resources
            rs.close();
            stmt.close();
            con.close();

        } catch (Exception e) {
            out.println("Error: " + e.getMessage());
        }
    }
}
